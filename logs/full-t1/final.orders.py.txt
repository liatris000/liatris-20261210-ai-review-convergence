"""Small order-handling helpers used as the review subject."""

import json


def parse_quantity(text):
    """Parse a quantity string such as "3" into a positive int.

    Raises ValueError with a descriptive message if the text is not a
    whole number or if the quantity is not positive.
    """
    stripped = str(text).strip()
    try:
        quantity = int(stripped)
    except ValueError:
        raise ValueError(f"Invalid quantity: {text!r} is not a whole number") from None
    if quantity <= 0:
        raise ValueError(f"Invalid quantity: {quantity} must be greater than zero")
    return quantity


def apply_discount(total, percent):
    """Return the total after a percentage discount.

    Raises ValueError if percent is not within 0-100.
    """
    if not 0 <= percent <= 100:
        raise ValueError(f"Invalid discount: {percent} must be between 0 and 100")
    return total - total * percent / 100


def paginate(items, page, size):
    """Return the items on a 1-indexed page.

    Raises ValueError if page or size is less than 1.
    """
    if page < 1:
        raise ValueError(f"Invalid page: {page} must be 1 or greater")
    if size < 1:
        raise ValueError(f"Invalid page size: {size} must be 1 or greater")
    start = (page - 1) * size
    return items[start:start + size]


def merge_tags(new_tags, tags=None):
    """Add new tags to the list, skipping duplicates."""
    if tags is None:
        tags = []
    for tag in new_tags:
        if tag not in tags:
            tags.append(tag)
    return tags


def load_config(path):
    """Load a JSON config file. Missing files give an empty config.

    Malformed JSON and permission errors are propagated to the caller.
    """
    try:
        with open(path) as f:
            return json.load(f)
    except FileNotFoundError:
        return {}


def average_price(orders):
    """Return the quantity-weighted average unit price across orders.

    Each order contributes its price once per unit ordered, so the result
    is total revenue divided by total units. Raises ValueError if there
    are no orders or the total quantity is zero.
    """
    if not orders:
        raise ValueError("Cannot compute average price of an empty order list")
    total_qty = sum(o["qty"] for o in orders)
    if total_qty == 0:
        raise ValueError("Cannot compute average price when total quantity is zero")
    total = sum(o["price"] * o["qty"] for o in orders)
    return total / total_qty


def order_summary(orders):
    """Return a one-line summary of the orders."""
    count = len(orders)
    total = sum(o["price"] * o["qty"] for o in orders)
    return f"{count} orders, total {total}"
